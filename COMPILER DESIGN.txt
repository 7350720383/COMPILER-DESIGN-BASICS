#include <iostream>
#include <sstream>
#include <cctype>

using namespace std;

class Parser {
    string input;
    size_t pos;

public:
    Parser(const string &expr) : input(expr), pos(0) {}

    char peek() {
        while (pos < input.size() && isspace(input[pos])) pos++; // Skip whitespace
        return (pos < input.size()) ? input[pos] : '\0';
    }

    char get() {
        char ch = peek();
        pos++;
        return ch;
    }

    double number() {
        string num;
        while (isdigit(peek()) || peek() == '.') {
            num += get();
        }
        return stod(num);
    }

    double factor() {
        if (peek() == '(') {
            get(); // Consume '('
            double result = expression();
            get(); // Consume ')'
            return result;
        } else {
            return number();
        }
    }

    double term() {
        double result = factor();
        while (peek() == '*' || peek() == '/') {
            char op = get();
            double right = factor();
            if (op == '*') result *= right;
            else if (op == '/') result /= right;
        }
        return result;
    }

    double expression() {
        double result = term();
        while (peek() == '+' || peek() == '-') {
            char op = get();
            double right = term();
            if (op == '+') result += right;
            else if (op == '-') result -= right;
        }
        return result;
    }
};

int main() {
    string expr;
    cout << "Enter an arithmetic expression: ";
    getline(cin, expr);
    
    Parser parser(expr);
    double result = parser.expression();
    cout << "Result: " << result << endl;
    
    return 0;
}
